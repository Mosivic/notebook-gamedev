1. Audio Dock中新建两个Bus, 取名为Spectrum和Record,分别加上效果器SpectrumAnalyzer和Record, 保存为资源文件
2.  树形结构为Control/Timer+AudioStreamPlayer
	1. 将 AudioStreamPlayer Bus设置为Spectrum, 即将信号传输给此Bus
	2. AudioStreamPlayer 创建麦克风Stream
3. 将项目设置中 Enable Audio Input 选项打开, 即可以接收外部麦克风
	Borderles打开, Energy Saving打开, Per Pixel Transparency打开
	Alaways on top打开, 调整分辨率



* 代码:
``` js
extends Control

const VU_CUT = 16
const FRQMAX = 12000.0
const w = 500
const h = 120
const MINDB =60

var spectrum :AudioEffectInstance
## 自定义可拖拽操作
var is_can_drag = false
var start_pos:Vector2
var offset:Vector2

var d_color = Color.green

func _ready():
	spectrum = AudioServer.get_bus_effect_instance(1,0) #第二个bus,第一个effect
	OS.set_window_always_on_top(true) #设置总是在前
	#get_tree().get_root().set_transparent_background(true) #设置背景透明
	
	
func _on_Timer_timeout():
	update() #时间一到更新绘制


func _draw():
	var colw = w/VU_CUT
	var preHZ = 0
	for i in range(1,VU_CUT+1):
		var hz = i * FRQMAX/VU_CUT
		# 当没有音频输入时 mag 的值恒为0, 不绘制图像
		var mag:float = spectrum.get_magnitude_for_frequency_range(preHZ,hz).length()
		var energy = clamp((MINDB+linear2db(mag))/MINDB,0,1)
		var colh = energy * h
		var c_color = (d_color if energy < 5 else Color.royalblue)*(.3+energy/3*2)
		draw_rect(Rect2(colw*i,h-colh,colh,colh+2),Color.green)
		preHZ = hz
```
