## 移动
### Topdown移动
#### 八向移动
``` js
extends KinematicBody2D

var speed = 200  # speed in pixels/sec
var velocity = Vector2.ZERO

func get_input():
    velocity = Vector2.ZERO
    if Input.is_action_pressed('right'):
        velocity.x += 1
    if Input.is_action_pressed('left'):
        velocity.x -= 1
    if Input.is_action_pressed('down'):
        velocity.y += 1
    if Input.is_action_pressed('up'):
        velocity.y -= 1
    # Make sure diagonal movement isn't faster
    velocity = velocity.normalized() * speed

func _physics_process(delta):
    get_input()
    velocity = move_and_slide(velocity)

```


#### 旋转与移动
``` js
extends KinematicBody2D

var speed = 200
var rotation_speed = 1.5

var velocity = Vector2.ZERO
var rotation_dir = 0

func get_input():
    rotation_dir = 0
    velocity = Vector2.ZERO
    if Input.is_action_pressed('right'):
        rotation_dir += 1
    if Input.is_action_pressed('left'):
        rotation_dir -= 1
    if Input.is_action_pressed('down'):
        velocity -= transform.x * speed
    if Input.is_action_pressed('up'):
        velocity += transform.x * speed

func _physics_process(delta):
    get_input()
    rotation += rotation_dir * rotation_speed * delta
    velocity = move_and_slide(velocity)

```

####  朝鼠标方向移动
``` js
extends KinematicBody2D

var speed = 200

var velocity = Vector2.ZERO

func get_input():
    velocity = Vector2.ZERO
    if Input.is_action_pressed("forward"):
        velocity += transform.x * speed
    if Input.is_action_pressed("back"):
        velocity -= transform.x * speed

func _physics_process(delta):
    look_at(get_global_mouse_position())
    get_input()
    velocity = move_and_slide(velocity)

```

#### 点击移动
``` js
extends KinematicBody2D

var speed = 200
var target = null
var velocity = Vector2.ZERO

func _input(event):
    if event.is_action_pressed('click'):
        target = event.position

func _physics_process(delta):
    if target:
        look_at(target)
        velocity = transform.x * speed
        # stop moving if we get close to the target
        if position.distance_to(target) > 5:
            velocity = move_and_slide(velocity)

```

#### 插值移动
``` js
extends KinematicBody2D

export var speed = 200
export var friction = 0.01
export var acceleration = 0.1

var velocity = Vector2()

func get_input():
    var input = Vector2()
    if Input.is_action_pressed('right'):
        input.x += 1
    if Input.is_action_pressed('left'):
        input.x -= 1
    if Input.is_action_pressed('down'):
        input.y += 1
    if Input.is_action_pressed('up'):
        input.y -= 1
    return input

func _physics_process(delta):
    var direction = get_input()
    if direction.length() > 0:
        velocity = lerp(velocity, direction.normalized() * speed, acceleration)
    else:
        velocity = lerp(velocity, Vector2.ZERO, friction)
    velocity = move_and_slide(velocity)

```

### 屏幕震动
``` js
extends Camera2D

export var decay = 0.8  # 多久屏幕震动停止 [0, 1].
export var max_offset = Vector2(100, 75)  # 最大的震动偏移像素.
export var max_roll = 0.1  # 最大的角度旋转.
export (NodePath) var target  # 相机跟随目标.

var trauma = 0.0  # 当前屏幕创动强度
var trauma_power = 2  # 创动指数. Use [2, 3].


func _ready():
    randomize()

# 增强创伤
func add_trauma(amount):
    trauma = min(trauma + amount, 1.0)

# 创伤逐渐减少
func _process(delta):
    if target:
        global_position = get_node(target).global_position
    if trauma:
        trauma = max(trauma - decay * delta, 0)
        shake()

# Shake() 函数根据当前的创伤程度（提升到给定的功率）对相机应用随机运动。在 2D 中，我们需要# 计算两个平移和一个旋转运动。在 -1 和 1 之间选择一个随机数是一种在任一方向上移动的简单方
# 法。

func shake():
    var amount = pow(trauma, trauma_power)
    rotation = max_roll * amount * rand_range(-1, 1)
    offset.x = max_offset.x * amount * rand_range(-1, 1)
    offset.y = max_offset.y * amount * rand_range(-1, 1)

# 使用 rand_range() 进行运动的缺点是它可以在帧之间发生很大变化，从而产生相当不稳定的感觉。# 为了获得更“平滑”的随机性，我们可以求助于一种叫做噪声的东西。
# 噪声，或者更具体地说是梯度噪声，是一种生成更自然的“随机”模式的方法。为此，Godot 提供了 
# OpenSimplexNoise 类。
# OpenSimplexNoise 通过生成点的 3D“云”来工作。每个点的值范围从 -1 到 1。下面是 
# OpenSimplexNoise 生成的两个噪声示例。在图像中，每个像素的白色值都映射到该点的噪声值。

# 更改后:
onready var noise = OpenSimplexNoise.new()
var noise_y = 0

func _ready():
	randomize()
	noise.seed = randi()
	noise.period = 4
	noise.octaves = 2

func shake():
	noise_y += 1
	rotation = max_roll * amount * noise.get_noise_2d(noise.seed, noise_y)
	offset.x = max_offset.x * amount * noise.get_noise_2d(noise.seed*2, noise_y)
	offset.y = max_offset.y * amount * noise.get_noise_2d(noise.seed*3, noise_y)

# get_noise_2d() 返回给定 (x, y) 坐标处的噪声值。我们不想对所有三个偏移使用相同的噪声值，# 因此我们在噪声中任意选择三个不同（但距离较远）的 x 值，然后使用稳步增加的 noise_y 值沿梯# 度“移动”每个点

```

### 屏幕包围
` 类似于贪吃蛇可以从屏幕末端到屏幕另一端 `
``` js
# player.gd
# 获取屏幕大小
onready var screen_size = get_viewport_rect().size

# 比较玩家位置

if position.x > screen_size.x:
    position.x = 0
if position.x < 0:
    position.x = screen_size.x
if position.y > screen_size.y:
    position.y = 0
if position.y < 0:
    position.y = screen_size.y

# 主要实现方法: 使用wrapf函数循环值
position.x = wrapf(position.x, 0, screen_size.x)
position.y = wrapf(position.y, 0, screen_size.y)

```

## Tilemap
### 检测碰撞
当 KinematicBody2D 发生碰撞时，碰撞数据会在 KinematicCollision2D 对象中返回。 TileMap 充当单个对撞机，因此如果您引用 collider 属性，它将是 TileMap 节点。然后，您需要找出 TileMap 中的哪个图块位于碰撞位置。
假设您已获得存储在变量碰撞中的 KinematicCollision2D 对象：
``` js
# Confirm the colliding body is a TileMap
if collision.collider is TileMap:
    # Find the character's position in tile coordinates
    var tile_pos = collision.collider.world_to_map(position)
    # Find the colliding tile position
    tile_pos -= collision.normal
    # 获取tile ID
    var tile_id = collision.collider.get_cellv(tile_pos)

	# 获取tile名称
	var tile_name = collision.collider.tile_set.tile_get_name(tile_id)

	# 也可以设置该位置新ID
	collision.collider.set_cellv(tile_pos, new_id)

```
